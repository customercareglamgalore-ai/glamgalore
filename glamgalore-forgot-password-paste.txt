import crypto from 'crypto';

// Password-reset links are signed, self-contained tokens (no database table
// needed): customerId + expiry + a fingerprint of the customer's CURRENT
// password hash, all signed with SESSION_SECRET. Because the fingerprint is
// part of the token, a link stops working the moment the password changes —
// so each link can only be used once — and it expires after an hour anyway.

const RESET_TTL_MS = 60 * 60 * 1000;

function secret(): string {
  const s = process.env.SESSION_SECRET;
  if (!s) throw new Error('SESSION_SECRET is not set');
  return s;
}

function fingerprint(passwordHash: string): string {
  return crypto.createHash('sha256').update(passwordHash).digest('hex').slice(0, 16);
}

function sign(payload: string): string {
  return crypto.createHmac('sha256', secret()).update(`pw-reset:${payload}`).digest('hex');
}

export function createResetToken(customerId: string, passwordHash: string): string {
  const payload = `${customerId}.${Date.now() + RESET_TTL_MS}.${fingerprint(passwordHash)}`;
  return Buffer.from(`${payload}.${sign(payload)}`).toString('base64url');
}

// Returns the customer id if the token is genuine, unexpired and still matches
// the customer's current password hash; otherwise null.
export function verifyResetToken(token: string, currentPasswordHash: (customerId: string) => Promise<string | null>) {
  return (async () => {
    let decoded: string;
    try {
      decoded = Buffer.from(token, 'base64url').toString('utf8');
    } catch {
      return null;
    }
    const parts = decoded.split('.');
    if (parts.length !== 4) return null;
    const [customerId, expiresRaw, fp, signature] = parts;
    const payload = `${customerId}.${expiresRaw}.${fp}`;
    const a = Buffer.from(signature);
    const b = Buffer.from(sign(payload));
    if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) return null;
    if (!Number(expiresRaw) || Date.now() > Number(expiresRaw)) return null;
    const hash = await currentPasswordHash(customerId);
    if (!hash || fingerprint(hash) !== fp) return null;
    return customerId;
  })();
}

// Tiny in-memory limiter (per server process) — enough to stop someone using
// the "forgot password" form to flood an inbox. Resets if the server restarts.
const hits = new Map<string, number[]>();

export function allow(key: string, max: number, windowMs: number): boolean {
  const now = Date.now();
  const recent = (hits.get(key) || []).filter((t) => now - t < windowMs);
  if (recent.length >= max) {
    hits.set(key, recent);
    return false;
  }
  recent.push(now);
  hits.set(key, recent);
  if (hits.size > 5000) {
    for (const [k, v] of hits) if (v.every((t) => now - t >= windowMs)) hits.delete(k);
  }
  return true;
}

import { prisma } from '@/lib/db';

import { createResetToken } from '@/lib/password-reset';

const schema = z.object({ identifier: z.string().trim().min(1).max(200) });

const SITE_URL = process.env.NEXT_PUBLIC_SITE_URL || 'https://glamgalore.in';
const MAX_ACCOUNTS_PER_REQUEST = 3;

// Same matching rules as login: an email address, or a mobile number compared
// on its last 10 digits.
async function findAccounts(identifier: string) {
  if (identifier.includes('@')) {
    return prisma.customer.findMany({
      where: { email: { equals: identifier, mode: 'insensitive' } },
      take: 1,
    });
  }
  const digits = identifier.replace(/\D/g, '');
  if (digits.length < 10) return [];
  const last10 = digits.slice(-10);
  const rows = await prisma.$queryRaw<{ id: string }[]>(Prisma.sql`
    SELECT id FROM "Customer"
    WHERE phone IS NOT NULL
      AND right(regexp_replace(phone, '[^0-9]', '', 'g'), 10) = ${last10}
    LIMIT ${MAX_ACCOUNTS_PER_REQUEST}
  `);
  if (rows.length === 0) return [];
  return prisma.customer.findMany({ where: { id: { in: rows.map((r) => r.id) } } });
}

// Always answers the same way, whether or not an account exists, so the form
// can't be used to find out who has an account.
export async function POST(req: NextRequest) {
  const parsed = schema.safeParse(await req.json().catch(() => null));
  if (!parsed.success) {
    return NextResponse.json({ error: 'Please enter your email or mobile number' }, { status: 400 });
  }
  const { identifier } = parsed.data;

  const ip = req.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || 'unknown';
  const withinLimits =
    allow(`forgot-ip:${ip}`, 10, 60 * 60 * 1000) && allow(`forgot-id:${identifier.toLowerCase()}`, 3, 60 * 60 * 1000);
  if (!withinLimits) {
    return NextResponse.json({ error: 'Too many attempts. Please try again in an hour.' }, { status: 429 });
  }

  try {
    const accounts = await findAccounts(identifier);
    for (const account of accounts) {
      const token = createResetToken(account.id, account.passwordHash);
      const resetUrl = `${SITE_URL}/account/reset-password?token=${encodeURIComponent(token)}`;
      const { subject, html, text } = buildPasswordResetEmail(account.name, resetUrl);
      const sent = await sendMail({ to: account.email, subject, html, text });
      if (!sent && process.env.NODE_ENV !== 'production') {
        console.log('[dev] email not sent (SMTP not configured). Reset link for ' + account.email + ':\n' + resetUrl);
      }
    }
  } catch (err) {
    console.error('forgot-password failed:', err);
  }
  return NextResponse.json({ ok: true });
}

import { NextRequest, NextResponse } from 'next/server';
import { Prisma } from '@prisma/client';
import { z } from 'zod';
import { prisma } from '@/lib/db';
import { verifyPassword, setSessionCookie } from '@/lib/auth';

// Log in with either an email address or a mobile number. `email` is still
// accepted as the field name so any older cached copy of the login page keeps
// working.
const loginSchema = z.object({
  identifier: z.string().trim().min(1).max(200).optional(),
  email: z.string().trim().min(1).max(200).optional(),
  password: z.string().min(1).max(200),
});

const MAX_PHONE_CANDIDATES = 5;

// Phone numbers are stored exactly as the customer typed them ("+91 97697
// 08781", "09769708781", "9769708781"), and the column isn't unique. Compare
// on the last 10 digits instead, then let the password decide which of any
// matching accounts is the right one.
async function findByPhone(identifier: string) {
  const digits = identifier.replace(/\D/g, '');
  if (digits.length < 10) return [];
  const last10 = digits.slice(-10);

  const rows = await prisma.$queryRaw<{ id: string }[]>(Prisma.sql`
    SELECT id FROM "Customer"
    WHERE phone IS NOT NULL
      AND right(regexp_replace(phone, '[^0-9]', '', 'g'), 10) = ${last10}
    LIMIT ${MAX_PHONE_CANDIDATES}
  `);
  if (rows.length === 0) return [];
  return prisma.customer.findMany({ where: { id: { in: rows.map((r) => r.id) } } });
}

export async function POST(req: NextRequest) {
  const parsed = loginSchema.safeParse(await req.json().catch(() => null));
  const identifier = parsed.success ? parsed.data.identifier ?? parsed.data.email : undefined;
  if (!parsed.success || !identifier) {
    return NextResponse.json({ error: 'Invalid credentials' }, { status: 400 });
  }
  const { password } = parsed.data;

  const candidates = identifier.includes('@')
    ? await prisma.customer.findMany({
        where: { email: { equals: identifier, mode: 'insensitive' } },
        take: 1,
      })
    : await findByPhone(identifier);

  let matched: (typeof candidates)[number] | null = null;
  for (const candidate of candidates) {
    if (await verifyPassword(password, candidate.passwordHash)) {
      matched = candidate;
      break;
    }
  }
  if (!matched) {
    return NextResponse.json({ error: 'Invalid email, mobile number or password' }, { status: 401 });
  }

  await setSessionCookie(matched.id);

  return NextResponse.json({ customer: { id: matched.id, name: matched.name, email: matched.email } });
}

      <h1 className="section-heading">
        {mode === 'login' ? 'Log in' : mode === 'signup' ? 'Create an account' : 'Forgot your password?'}
      </h1>

      <form onSubmit={handleSubmit}>
        {error && <p style={{ color: '#e22120' }}>{error}</p>}
        {notice && <p style={{ color: '#2e7d32' }}>{notice}</p>}

        {mode === 'forgot' && (
          <p style={{ margin: '0 0 8px' }}>
            Enter the email or mobile number on your account and we&apos;ll email you a link to choose a new password.
          </p>
        )}

        {mode === 'signup' && (
          <input
            placeholder="Full name"
            value={form.name}
            onChange={(e) => setForm({ ...form, name: e.target.value })}
            required
            style={inputStyle}
          />
        )}
        {mode === 'login' || mode === 'forgot' ? (
          <input
            type="text"
            inputMode="email"
            autoComplete="username"
            autoCapitalize="none"
            autoCorrect="off"
            spellCheck={false}
            placeholder="Email or mobile number"
            aria-label="Email or mobile number"
            value={form.identifier}
            onChange={(e) => setForm({ ...form, identifier: e.target.value })}
            required
            style={inputStyle}
          />
        ) : (
          <input
            type="email"
            placeholder="Email"
            value={form.email}
            onChange={(e) => setForm({ ...form, email: e.target.value })}
            required
            style={inputStyle}
          />
        )}
        {mode === 'signup' && (
          <input
            type="tel"
            autoComplete="tel"
            placeholder="Mobile number (you can log in with this too)"
            value={form.phone}
            onChange={(e) => setForm({ ...form, phone: e.target.value })}
            style={inputStyle}
          />
        )}
        {mode !== 'forgot' && (
          <input
            type="password"
            placeholder="Password"
            value={form.password}
            onChange={(e) => setForm({ ...form, password: e.target.value })}
            required
            minLength={8}
            style={inputStyle}
          />
        )}

        <button type="submit" className="btn" disabled={submitting} style={{ width: '100%' }}>
          {submitting
            ? 'Please wait...'
            : mode === 'login'
              ? 'Log in'
              : mode === 'signup'
                ? 'Sign up'
                : 'Email me a reset link'}
        </button>
      </form>

      {mode === 'login' && (
        <p style={{ marginTop: 12 }}>
          <button type="button" style={linkStyle} onClick={() => switchMode('forgot')}>Forgot your password?</button>
        </p>
      )}

      <p style={{ marginTop: 16 }}>
        {mode === 'login' ? (
          <>Don&apos;t have an account?{' '}
            <button type="button" style={linkStyle} onClick={() => switchMode('signup')}>Sign up</button>
          </>
        ) : mode === 'signup' ? (
          <>Already have an account?{' '}
            <button type="button" style={linkStyle} onClick={() => switchMode('login')}>Log in</button>
          </>
        ) : (
          <>
            <button type="button" style={linkStyle} onClick={() => switchMode('login')}>Back to log in</button>
          </>
        )}
      </p>
      </Reveal>
    </main>
  );
}
